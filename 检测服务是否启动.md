1. TCP 端口检测（简单，但不完整）
判断应用端口是否监听：  nc -z 127.0.0.1 8080  或者： timeout 3 bash -c '</dev/tcp/127.0.0.1/8080'
wait_port()
{
    host=$1
    port=$2

    for i in $(seq 1 60)
    do
        if nc -z ${host} ${port}; then
            echo "${host}:${port} ready"
            return 0
        fi

        echo "waiting ${host}:${port}"
        sleep 2
    done

    return 1
}


wait_port 127.0.0.1 8080

5. 文件 Ready 标记（容器非常常用）
应用启动完成后创建：
/runtime/apps/app1/app.ready
Spring Boot：
@Component
public class ReadyFile {

    @EventListener(ApplicationReadyEvent.class)
    public void ready()
            throws Exception {

        Files.createFile(
          Path.of("/runtime/apps/app1/app.ready")
        );
    }
}
循环判断文件
while [ ! -f /runtime/apps/app1/app.ready ]
do
    sleep 2
done

增加等待函数
#!/bin/bash

set -e


wait_for_http()
{
    name=$1
    url=$2
    timeout=$3

    echo "[wait] waiting ${name} ..."

    for i in $(seq 1 ${timeout})
    do

        if curl -fs ${url} >/dev/null 2>&1
        then
            echo "[ok] ${name} is ready"
            return 0
        fi

        echo "[wait] ${name} not ready (${i}/${timeout})"

        sleep 2
    done


    echo "[error] ${name} start timeout"

    return 1
}
等待 app1：
wait_for_http \
"app1" \
"http://127.0.0.1:8080/actuator/health" \
60

wait_port()
{
    host=$1
    port=$2

    echo "waiting $host:$port"

    while ! nc -z $host $port

        sleep 2
    done

    echo "$host:$port ready"
}
